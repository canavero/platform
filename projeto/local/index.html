    Execução local

    Tudo roda na máquina com Docker — inclusive o build, se não houver Java instalado: o scripts/build.sh executa o Maven Wrapper (./mvnw) de cada módulo dentro de um container eclipse-temurin:25-jdk, um módulo por vez (interfaces primeiro, com install; depois os serviços, com package, que roda os testes).

    git clone --recurse-submodules https://github.com/EuMiguelll/Projeto-PMA.git && cd Projeto-PMA
./scripts/build.sh                 # 7 módulos: compila, testa e gera api/*-service/target/*.jar
docker compose up -d --build       # db, redis, account, auth, exchange, gateway, prometheus, grafana
./scripts/smoke-test.sh            # teste ponta a ponta pelo gateway
docker compose down                # (-v para apagar o volume do Postgres)
    URL O quê
    http://localhost:8080/info gateway (única porta da API exposta)
    http://localhost:9090 Prometheus (Status → Targets)
    http://localhost:3000 Grafana (admin/admin) → dashboard Store — Bottlenecks
    compose.yaml 
    # Plataforma completa local: docker compose up -d --build   (antes: ./scripts/build.sh)
# Só o gateway (8080) é exposto; os demais ficam na rede interna (trusted layer).
name: store

x-java: &java
  restart: unless-stopped
  deploy:
    resources:
      limits:
        memory: 512M

services:

  db:
    image: postgres:17
    environment:
      POSTGRES_USER: ${DB_USER:-store}
      POSTGRES_PASSWORD: ${DB_PASSWORD:-store}
      POSTGRES_DB: ${DB_NAME:-store}
    volumes:
      - db_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${DB_USER:-store}"]
      interval: 5s
      retries: 20

  redis: # cache (account, exchange) + rate limiting (gateway)
    image: redis:7-alpine

  account:
    <<: *java
    build: ./api/account-service
    environment:
      DATABASE_HOST: db
      DATABASE_PORT: 5432
      DATABASE_DB: ${DB_NAME:-store}
      DATABASE_USERNAME: ${DB_USER:-store}
      DATABASE_PASSWORD: ${DB_PASSWORD:-store}
      REDIS_HOST: redis
    depends_on:
      db:
        condition: service_healthy

  auth:
    <<: *java
    build: ./api/auth-service
    environment:
      JWT_SECRET_KEY: ${JWT_SECRET_KEY:-EBwrdOOcAIDRLON9hbJV0+O3PTdfTjgMCQIxwLH7ws8=}

  exchange:
    <<: *java
    build: ./api/exchange-service
    environment:
      REDIS_HOST: redis
      # demo do circuit breaker: EXCHANGE_PROVIDER_URL=http://redis:6380 docker compose up -d exchange
      EXCHANGE_PROVIDER_URL: ${EXCHANGE_PROVIDER_URL:-https://economia.awesomeapi.com.br}
      EXCHANGE_PROVIDER_TOKEN: ${EXCHANGE_PROVIDER_TOKEN:-} # API key da AwesomeAPI (opcional)

  # E2E no CI: substitui a AwesomeAPI (sem API key, IPs de nuvem recebem 429)
  #   EXCHANGE_PROVIDER_URL=http://provider-stub docker compose --profile e2e up -d
  provider-stub:
    image: nginx:1.29-alpine
    profiles: [e2e]
    volumes:
      - ./setup/provider-stub/default.conf:/etc/nginx/conf.d/default.conf:ro

  gateway:
    <<: *java
    build: ./api/gateway-service
    ports:
      - 8080:8080
    environment:
      REDIS_HOST: redis
      RATE_LIMIT_REPLENISH: ${RATE_LIMIT_REPLENISH:-50}
      RATE_LIMIT_BURST: ${RATE_LIMIT_BURST:-100}
    depends_on:
      - account
      - auth
      - exchange
      - redis

  # bottleneck: observabilidade
  prometheus:
    image: prom/prometheus:v3.5.0
    ports:
      - 9090:9090
    volumes:
      - ./setup/prometheus/prometheus.yml:/etc/prometheus/prometheus.yml:ro

  grafana:
    image: grafana/grafana:12.1.0
    ports:
      - 3000:3000
    environment:
      GF_SECURITY_ADMIN_PASSWORD: ${GRAFANA_PASSWORD:-admin}
    volumes:
      - ./setup/grafana/provisioning:/etc/grafana/provisioning:ro
    depends_on:
      - prometheus

volumes:
  db_data:

    Como no hands-on 2, só o gateway publica porta: account, auth, exchange, db e redis ficam na rede interna do compose (trusted layer). Os serviços Java têm limite de 512 MiB e a JVM usa -XX:MaxRAMPercentage=75.

    No Mac o script respeita a carga da máquina: só compila com load average < 5 e pressão de memória normal, com prioridade baixa (taskpolicy -b), -T 2 e no máximo 2 CPUs / 3 GB para o container do Maven.
