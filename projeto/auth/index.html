    Auth

    Repositórios: auth (interface) · auth-service · pacote store.auth

    Autenticação e autorização (hands-on 3): registra contas (delegando ao account), faz login emitindo um JWT HS256 em cookie HttpOnly, e resolve tokens para o gateway. Nunca vê o hash da senha: quem confere a senha é o account (POST /accounts/login).

    Método Caminho Descrição Respostas
    POST /auth/register cria a conta via Feign no account 200, 400, 409, 503
    POST /auth/login valida credenciais e devolve Set-Cookie: __store_jwt_token=<JWT> 200, 401, 503
    GET /auth/me conta logada (header id-account colocado pelo gateway) 200, 404
    POST /auth/solve interno: {token} → {idAccount} (usado pelo gateway) 200, 401
    GET /auth/logout apaga o cookie (Max-Age=0) 200

    Claims do token: jti = id da conta, sub = nome, email, iss = Insper::PMA, nbf, exp (1 h). A chave vem de JWT_SECRET_KEY (Base64, 256 bits; o deploy.sh gera uma aleatória no Kubernetes — a de desenvolvimento do application.yaml é pública).

    package store.auth;

import java.util.Map;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.http.HttpStatusCode;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

import feign.FeignException;
import io.github.resilience4j.circuitbreaker.annotation.CircuitBreaker;
import store.account.AccountController;
import store.account.AccountIn;
import store.account.AccountOut;

@Service
public class AuthService {

    @Autowired
    private AccountController accountController;

    @Autowired
    private JwtService jwtService;

    @Value("${store.jwt.duration}")
    private Long duration;

    @Value("${store.jwt.httpOnly}")
    private Boolean httpOnly;

    // bottleneck (circuit breaker): se o account cair, falhamos rápido com 503
    // em vez de segurar threads esperando timeout (ver resilience4j em application.yaml)
    @CircuitBreaker(name = "account", fallbackMethod = "createFallback")
    public void create(Register register) {
        accountController.create(AccountIn.builder()
            .name(register.name())
            .email(register.email())
            .password(register.password())
            .build());
    }

    @CircuitBreaker(name = "account", fallbackMethod = "loginFallback")
    public String login(String email, String password) {
        final AccountOut account;
        try {
            // o account calcula o hash da senha e compara com o armazenado
            account = accountController.login(AccountIn.builder().email(email).password(password).build()).getBody();
        } catch (FeignException.Unauthorized | FeignException.BadRequest e) {
            throw new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Invalid credentials");
        }
        return jwtService.generate(account, getDuration());
    }

    @CircuitBreaker(name = "account", fallbackMethod = "meFallback")
    public AccountOut me(String idAccount) {
        return accountController.findById(idAccount).getBody();
    }

    public Map<String, String> solve(String jwt) {
        return Map.of("idAccount", jwtService.getId(jwt));
    }

    public long getDuration() {
        return duration;
    }

    public boolean getHttpOnly() {
        return httpOnly;
    }

    private void createFallback(Register register, RuntimeException e) { throw translate(e); }
    private String loginFallback(String email, String password, RuntimeException e) { throw translate(e); }
    private AccountOut meFallback(String idAccount, RuntimeException e) { throw translate(e); }

    // 4xx do account passa adiante (400, 404, 409); falha de rede/5xx ou circuito aberto vira 503
    static RuntimeException translate(RuntimeException e) {
        if (e instanceof ResponseStatusException) return e;
        if (e instanceof FeignException f && f.status() >= 400 && f.status() < 500) {
            return new ResponseStatusException(HttpStatusCode.valueOf(f.status()));
        }
        return new ResponseStatusException(HttpStatus.SERVICE_UNAVAILABLE, "account service unavailable", e);
    }

}

    package store.auth;

import java.util.Base64;
import java.util.Date;
import java.util.Map;

import javax.crypto.SecretKey;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.HttpStatus;
import org.springframework.stereotype.Service;
import org.springframework.web.server.ResponseStatusException;

import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import store.account.AccountOut;

@Service
public class JwtService {

    @Value("${store.jwt.secretKey}")
    private String secretKey;

    public String generate(AccountOut account, long duration) {
        Date now = new Date();
        return Jwts.builder()
            .id(account.id())
            .issuer("Insper::PMA")
            .claims(Map.of("email", account.email()))
            .subject(account.name())
            .notBefore(now)
            .expiration(new Date(now.getTime() + duration))
            .signWith(getKey())
            .compact();
    }

    // valida assinatura, nbf e exp; qualquer problema vira 401
    public String getId(String jwt) {
        try {
            return Jwts.parser().verifyWith(getKey()).build()
                .parseSignedClaims(jwt).getPayload().getId();
        } catch (JwtException | IllegalArgumentException e) {
            throw new ResponseStatusException(HttpStatus.UNAUTHORIZED, "Invalid token");
        }
    }

    private SecretKey getKey() {
        return Keys.hmacShaKeyFor(Base64.getDecoder().decode(secretKey));
    }

}

    package store.auth;

import java.util.Map;

import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;

import store.account.AccountOut;

@FeignClient(name = "auth", url = "${store.auth.url:http://auth:8080}")
public interface AuthController {

    String AUTH_COOKIE_TOKEN = "__store_jwt_token";

    @PostMapping("/auth/register")
    ResponseEntity<Void> create(@RequestBody RegisterIn in);

    @PostMapping("/auth/login")
    ResponseEntity<Void> login(@RequestBody LoginIn in);

    @GetMapping("/auth/me")
    ResponseEntity<AccountOut> me(@RequestHeader(value = "id-account", required = true) String idAccount);

    // endpoint interno: o gateway troca o JWT pelo id da conta
    @PostMapping("/auth/solve")
    ResponseEntity<Map<String, String>> solveToken(@RequestBody TokenOut token);

    @GetMapping("/auth/logout")
    ResponseEntity<Void> logout();

    @GetMapping("/auth/health-check")
    ResponseEntity<Void> healthCheck();

}
