    Exchange

    Repositórios: exchange (interface) · exchange-service · pacote store.exchange

    Cotações de câmbio (exercício Exchange API): GET /exchanges/{from}/{to} consulta a AwesomeAPI (gratuita, sem chave) e responde no formato pedido pelo enunciado. Exige usuário autenticado (o gateway injeta o id-account).

    GET /exchanges/USD/BRL
Cookie: __store_jwt_token=...

200 OK
{"sell":5.668,"buy":5.6666,"date":"2026-11-13 18:30:13","id-account":"7586833b-189d-4f47-a27d-0cf8589a8b3b"}

    sell = ask (preço de venda) e buy = bid (preço de compra) da AwesomeAPI. Par inválido (ex.: XXX/YYY) → 404; provedor fora do ar → 503.

    Por que Java e não Python?

    O exercício individual Exchange API pede FastAPI/Python. No projeto, o Exchange foi feito em Spring Boot, como os outros três serviços, para manter um único padrão (módulo de interface Feign exchange reutilizável pelo futuro order-service, mesmo pipeline de CI, mesmo Dockerfile, mesmas métricas/probes). O contrato HTTP é o do enunciado, então o serviço é intercambiável com uma versão FastAPI.

    package store.exchange;

import java.time.Duration;
import java.util.Map;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.core.ParameterizedTypeReference;
import org.springframework.http.HttpStatus;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.stereotype.Service;
import org.springframework.web.client.HttpClientErrorException;
import org.springframework.web.client.RestClient;
import org.springframework.web.server.ResponseStatusException;

import io.github.resilience4j.circuitbreaker.annotation.CircuitBreaker;
import io.github.resilience4j.retry.annotation.Retry;

@Service
public class ExchangeService {

    private final RestClient provider;

    @Autowired
    public ExchangeService(@Value("${store.exchange.provider}") String url,
                           @Value("${store.exchange.token:}") String token) {
        this(RestClient.builder().baseUrl(url).requestFactory(timeouts())
            .defaultHeaders(h -> { if (!token.isBlank()) h.set("x-api-key", token); })
            .build());
    }

    ExchangeService(RestClient provider) {
        this.provider = provider;
    }

    // bottlenecks: cache (60 s no Redis, compartilhado entre réplicas) evita bater na API
    // de terceiros a cada request; retry + circuit breaker protegem contra lentidão/queda dela
    @Cacheable(value = "exchanges", key = "#from + '-' + #to")
    @Retry(name = "provider", fallbackMethod = "unavailable")
    @CircuitBreaker(name = "provider")
    public Rate rate(String from, String to) {
        // AwesomeAPI: {"USDBRL": {"bid": "5.66", "ask": "5.67", "create_date": "2026-11-13 18:30:13", ...}}
        Map<String, Map<String, String>> body = provider.get()
            .uri("/json/last/{from}-{to}", from, to)
            .retrieve()
            .body(new ParameterizedTypeReference<>() {});
        Map<String, String> q = body.get(from + to);
        return new Rate(Double.valueOf(q.get("ask")), Double.valueOf(q.get("bid")), q.get("create_date"));
    }

    // fallback no @Retry (camada externa): roda uma vez, após as tentativas ou com o circuito aberto
    private Rate unavailable(String from, String to, Throwable e) {
        if (e instanceof HttpClientErrorException.NotFound) {
            throw new ResponseStatusException(HttpStatus.NOT_FOUND, "unknown currency pair " + from + "-" + to);
        }
        throw new ResponseStatusException(HttpStatus.SERVICE_UNAVAILABLE, "exchange provider unavailable");
    }

    private static SimpleClientHttpRequestFactory timeouts() {
        SimpleClientHttpRequestFactory f = new SimpleClientHttpRequestFactory();
        f.setConnectTimeout(Duration.ofSeconds(2));
        f.setReadTimeout(Duration.ofSeconds(3));
        return f;
    }

}

    package store.exchange;

import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.RestController;

@RestController
public class ExchangeResource implements ExchangeController {

    @Autowired
    private ExchangeService exchangeService;

    @Override
    public ResponseEntity<ExchangeOut> exchange(String from, String to, String idAccount) {
        Rate r = exchangeService.rate(from.toUpperCase(), to.toUpperCase());
        return ResponseEntity.ok(ExchangeOut.builder()
            .sell(r.sell())
            .buy(r.buy())
            .date(r.date())
            .idAccount(idAccount)
            .build());
    }

    @Override
    public ResponseEntity<Void> healthCheck() {
        return ResponseEntity.ok().build();
    }

}

    package store.exchange;

import org.springframework.cloud.openfeign.FeignClient;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestHeader;

@FeignClient(name = "exchange", url = "${store.exchange.url:http://exchange:8080}")
public interface ExchangeController {

    // ex.: GET /exchanges/USD/BRL
    @GetMapping("/exchanges/{from}/{to}")
    ResponseEntity<ExchangeOut> exchange(
        @PathVariable String from,
        @PathVariable String to,
        @RequestHeader(value = "id-account", required = true) String idAccount
    );

    @GetMapping("/exchanges/health-check")
    ResponseEntity<Void> healthCheck();

}
